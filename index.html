int line; }; // ============================ Lexer ============================= enum class T { Num, Str, Id, Let, Fn, If, Else, While, For, Return, True, False, Null, Break, Continue, LP, RP, LB, RB, Comma, Semi, Plus, Minus, Star, Slash, Percent, Bang, Assign, EqEq, Ne, Lt, Le, Gt, Ge, And, Or, End }; struct Token { T t; string s; double n = 0; int line = 1; }; vector lex(const string& src) { static const unordered_map kw = { {"let", T::Let}, {"fn", T::Fn}, {"if", T::If}, {"else", T::Else}, {"while", T::While}, {"for", T::For}, {"return", T::Return}, {"true", T::True}, {"false", T::False}, {"null", T::Null}, {"break", T::Break}, {"continue", T::Continue}}; i++; } else if (isspace((unsigned char)c)) i++; else if (c == '/' && i + 1 < n && src[i + 1] == '/') { while (i < n && src[i] != '\n') i++; } else if (isdigit((unsigned char)c)) { size_t j = i; while (j < n && isdigit((unsigned char)src[j])) j++; while (j < n && (isalnum((unsigned char)src[j]) || src[j] == '_')) j++; string w = src.substr(i, j - i); auto it = kw.find(w); add(it != kw.end() ? it->second : T::Id, w); i = j; } else if (c == '"') { string s; i++; while (i < n && src[i] != '"') { if (src[i] == '\n') line++; if (src[i] == '\\' && i + 1 < n) { i++; switch (src[i]) { case 'n': s += '\n'; break; case 't': s += '\t'; break; case '"': s += '"'; break; case '\\': s += '\\'; break; default: s += src[i]; } } else s += src[i]; i++; } if (i >= n) throw AcribeError{"Unterminated string", line}; i++; add(T::Str, s); } else { auto two = [&](char nx) { return i + 1 < n && src[i + 1] == nx; }; switch (c) { case '(': add(T::LP); break; case ')': add(T::RP); break; case '{': add(T::LB); break; case '}': add(T::RB); break; case ',': add(T::Comma); break; case ';': add(T::Semi); break; case '+': add(T::Plus); break; case '-': add(T::Minus); break; case '*': add(T::Star); break; case '/': add(T::Slash); break; case '%': add(T::Percent); break; case '!': if (two('=')) { add(T::Ne); i++; } else add(T::Bang); break; case '=': if (two('=')) { add(T::EqEq); i++; } else add(T::Assign); break; case '<': if (two('=')) { add(T::Le); i++; } else add(T::Lt); break; case '>': if (two('=')) { add(T::Ge); i++; } else add(T::Gt); break; case '&': if (two('&')) { add(T::And); i++; } else throw AcribeError{"Unexpected '&'", line}; break; case '|': if (two('|')) { add(T::Or); i++; } else throw AcribeError{"Unexpected '|'", line}; break; default: throw AcribeError{string("Unexpected character '") + c + "'", line}; } i++; } } add(T::End); return out; } // ============================ Values ============================ struct Interp; struct Callable; struct Value { enum Ty { Nil, Num, Bool, Str, Fn } ty = Nil; double n = 0; bool b = false; string s; shared_ptr f; static Value num(double d) { Value v; v.ty = Num; v.n = d; return v; } static Value boolean(bool x) { Value v; v.ty = Bool; v.b = x; return v; } static Value str(string x) { Value v; v.ty = Str; v.s = move(x); return v; } }; struct Callable { int arity = 0; string name; // arity -1 = variadic virtual ~Callable() {} virtual Value call(Interp&, vector&, int line) = 0; }; string numToStr(double d) { if (d == floor(d) && fabs(d) < 1e15) { char b[32]; snprintf(b, sizeof b, "%.0f", d); return b; } char b[40]; snprintf(b, sizeof b, "%.14g", d); return b; } string toStr(const Value& v) { switch (v.ty) { case Value::Nil: return "null"; case Value::Num: return numToStr(v.n); case Value::Bool: return v.b ? "true" : "false"; case Value::Str: return v.s; case Value::Fn: return "name + ">"; } return ""; } bool truthy(const Value& v) { return !(v.ty == Value::Nil || (v.ty == Value::Bool && !v.b)); } bool equals(const Value& a, const Value& b) { if (a.ty != b.ty) return false; } // ============================== AST ============================= struct Expr; struct Stmt; using EP = shared_ptr; using SP = shared_ptr; struct Expr { enum K { Lit, Var, Assign, Unary, Binary, Logic, Call } k; SP a, b; vector params; int line = 0; }; // ============================ Parser ============================ struct Parser { vector tk; size_t p = 0; explicit Parser(vector t) : tk(move(t)) {} const Token& cur() { return tk[p]; } bool check(T t) { return cur().t == t; } bool match(T t) { if (check(t)) { p++; return true; } return false; } const Token& expect(T t, const string& msg) { if (!check(t)) throw AcribeError{msg, cur().line}; expect(T::RB, "Expected '}'"); return v; } SP stmt() { int ln = cur().line; if (match(T::LB)) { auto s = mk(Stmt::Block, ln); s->body = blockBody(); return s; } if (match(T::If)) { auto s = mk(Stmt::If, ln); expect(T::LP, "Expected '(' after 'while'"); s->e = expr(); expect(T::RP, "Expected ')'"); s->a = stmt(); return s; } if (match(T::For)) { auto s = mk(Stmt::For, ln); expect(T::LP, "Expected '(' after 'for'"); expect(T::Semi, "Expected ';' after loop condition"); if (!check(T::RP)) s->inc = expr(); expect(T::RP, "Expected ')' after for clauses"); s->b = stmt(); return s; } if (match(T::Return)) { auto s = mk(Stmt::Return, ln); if (!check(T::Semi)) s->e = expr(); expect(T::Semi, "Expected ';' after return"); return s; } if (match(T::Break)) { expect(T::Semi, "Expected ';'"); return mk(Stmt::Break, ln); } if (match(T::Continue)) { expect(T::Semi, "Expected ';'"); return mk(Stmt::Continue, ln); } auto s = mk(Stmt::ExprS, ln); s->e = expr(); expect(T::Semi, "Expected ';' after expression"); return s; } EP expr() { return assignment(); } EP assignment() { EP l = orE(); if (check(T::Assign)) { int ln = cur().line; p++; EP v = assignment(); if (l->k != Expr::Var) throw AcribeError{"Invalid assignment target", ln}; auto e = mkE(Expr::Assign, ln); e->name = l->name; e->a = v; return e; } return l; } EP bin(Expr::K k, T op, EP a, EP b, int ln) { auto e = mkE(k, ln); e->op = op; e->a = a; e->b = b; return e; } EP orE() { EP l = andE(); while (check(T::Or)) { int ln = cur().line; p++; l = bin(Expr::Logic, T::Or, l, andE(), ln); } return l; } EP andE() { EP l = eq(); while (check(T::And)) { int ln = cur().line; p++; l = bin(Expr::Logic, T::And, l, eq(), ln); } return l; } EP eq() { EP l = cmp(); while (check(T::EqEq) || check(T::Ne)) { T o = cur().t; int ln = cur().line; p++; l = bin(Expr::Binary, o, l, cmp(), ln); } return l; } EP cmp() { EP l = term(); while (check(T::Lt) || check(T::Le) || check(T::Gt) || check(T::Ge)) { T o = cur().t; int ln = cur().line; p++; l = bin(Expr::Binary, o, l, term(), ln); } return l; } EP term() { EP l = factor(); while (check(T::Plus) || check(T::Minus)) { T o = cur().t; int ln = cur().line; p++; l = bin(Expr::Binary, o, l, factor(), ln); } return l; } EP factor() { EP l = unary(); while (check(T::Star) || check(T::Slash) || check(T::Percent)) { T o = cur().t; int ln = cur().line; p++; l = bin(Expr::Binary, o, l, unary(), ln); } return l; } EP unary() { if (check(T::Bang) || check(T::Minus)) { T o = cur().t; int ln = cur().line; p++; auto e = mkE(Expr::Unary, ln); e->op = o; e->a = unary(); return e; } return call(); } EP call() { EP e = primary(); while (check(T::LP)) { int ln = cur().line; p++; auto c = mkE(Expr::Call, ln); c->a = e; if (!check(T::RP)) do { c->args.push_back(expr()); } while (match(T::Comma)); expect(T::RP, "Expected ')' after arguments"); e = c; } return e; } EP primary() { const Token& t = cur(); int ln = t.line; if (match(T::Num)) { auto e = mkE(Expr::Lit, ln); e->val = Value::num(t.n); return e; } if (match(T::Str)) { auto e = mkE(Expr::Lit, ln); e->val = Value::str(t.s); return e; } if (match(T::True)) { auto e = mkE(Expr::Lit, ln); e->val = Value::boolean(true); return e; } if (match(T::False)) { auto e = mkE(Expr::Lit, ln); e->val = Value::boolean(false); return e; } if (match(T::Null)) { auto e = mkE(Expr::Lit, ln); return e; } if (match(T::Id)) { auto e = mkE(Expr::Var, ln); e->name = t.s; return e; } if (match(T::LP)) { EP e = expr(); expect(T::RP, "Expected ')'"); return e; } throw AcribeError{"Unexpected token in expression", ln}; } }; // =========================== Interpreter ======================== struct Env { unordered_map vars; shared_ptr parent; explicit Env(shared_ptr p = nullptr) : parent(move(p)) {} Value* find(const string& n) { for (Env* e = this;
